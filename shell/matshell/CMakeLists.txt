cmake_minimum_required(VERSION 3.16)
project(MaterializedShell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick DBus)
if(NOT TARGET matdbus)
    add_subdirectory(../../libs/libmatdbus ${CMAKE_BINARY_DIR}/libmatdbus)
endif()
add_executable(matshell main.cpp application_model.cpp application_model.hpp notification_server.cpp notification_server.hpp power_backend.cpp power_backend.hpp theme_controller.cpp theme_controller.hpp device_backend.cpp device_backend.hpp)
qt_add_resources(matshell shell_resources PREFIX /shell FILES Main.qml Panel.qml Launcher.qml QuickSettings.qml NotificationCenter.qml)
target_link_libraries(matshell PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::DBus matdbus)
