cmake_minimum_required(VERSION 3.16)
project(MaterializedSession LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus)
if(NOT TARGET m3)
    add_subdirectory(../../libs/libm3 ${CMAKE_BINARY_DIR}/libm3)
endif()
if(NOT TARGET matdbus)
    add_subdirectory(../../libs/libmatdbus ${CMAKE_BINARY_DIR}/libmatdbus)
endif()
add_executable(matsessiond main.cpp themed_service.cpp themed_service.hpp)
target_include_directories(matsessiond PRIVATE . ../../libs/libmatdbus)
target_link_libraries(matsessiond PRIVATE Qt6::Core Qt6::Gui Qt6::DBus m3 matdbus)
