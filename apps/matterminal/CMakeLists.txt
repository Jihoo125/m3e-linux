cmake_minimum_required(VERSION 3.16)
project(MaterializedTerminal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(matterminal main.cpp terminal_emulator.cpp terminal_emulator.hpp pty_session.cpp pty_session.hpp)
qt_add_resources(matterminal resources PREFIX /matterminal FILES Main.qml)
target_link_libraries(matterminal PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick util)
if(MAT_ENABLE_TESTS)
    enable_testing()
    add_executable(matterminal-tests terminal_test.cpp terminal_emulator.cpp terminal_emulator.hpp)
    target_link_libraries(matterminal-tests PRIVATE Qt6::Core)
    add_test(NAME matterminal COMMAND matterminal-tests)
endif()
