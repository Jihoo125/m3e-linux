cmake_minimum_required(VERSION 3.16)
project(MaterializedCalculator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(matcalc main.cpp calculator_engine.cpp calculator_engine.hpp)
qt_add_resources(matcalc resources PREFIX /matcalc FILES Main.qml)
target_link_libraries(matcalc PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
if(MAT_ENABLE_TESTS)
    enable_testing()
    add_executable(matcalc-tests calculator_test.cpp calculator_engine.cpp calculator_engine.hpp)
    target_link_libraries(matcalc-tests PRIVATE Qt6::Core)
    add_test(NAME matcalc COMMAND matcalc-tests)
endif()
