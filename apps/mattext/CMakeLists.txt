cmake_minimum_required(VERSION 3.16)
project(MaterializedText LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(mattext main.cpp text_document.cpp text_document.hpp)
qt_add_resources(mattext resources PREFIX /mattext FILES Main.qml)
target_link_libraries(mattext PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
if(MAT_ENABLE_TESTS)
    enable_testing()
    add_executable(mattext-tests document_test.cpp text_document.cpp text_document.hpp)
    target_link_libraries(mattext-tests PRIVATE Qt6::Core)
    add_test(NAME mattext COMMAND mattext-tests)
endif()
