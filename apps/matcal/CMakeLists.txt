cmake_minimum_required(VERSION 3.16)
project(MaterializedCalendar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(matcal main.cpp calendar_model.cpp calendar_model.hpp)
qt_add_resources(matcal resources PREFIX /matcal FILES Main.qml)
target_link_libraries(matcal PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
if(MAT_ENABLE_TESTS)
    enable_testing()
    add_executable(matcal-tests calendar_test.cpp calendar_model.cpp calendar_model.hpp)
    target_link_libraries(matcal-tests PRIVATE Qt6::Core)
    add_test(NAME matcal COMMAND matcal-tests)
endif()
